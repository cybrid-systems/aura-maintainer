(lambda (dgets dsets dhits dmisses devicted nkeys dexpired avg_ttl keys_ttl)
  (let ((ops (+ dgets dsets))
        (miss-pct (if (> (+ dhits dmisses) 0)
                    (quotient (* 100 dmisses) (+ dhits dmisses))
                    0))
        (get-set (if (> ops 0) (quotient (* 100 dgets) ops) 0))
        (erate (if (> ops 0)
                 (quotient (* 100 devicted) ops)
                 0)))
    (cond
      ((and (>= erate 16) (> dsets dgets))
       (if (>= erate 22) "lru|flat|soft" "lfu|flat|pin"))
      ((and (>= erate 16) (> dgets dsets))
       "lfu|flat")
      ((>= erate 25)
       (if (and (> keys_ttl 0) (< avg_ttl 12000))
         "ttl_aware|flat|soft"
         "lru|flat|soft"))
      ((and (> keys_ttl 0)
            (< avg_ttl 12000)
            (>= dgets 10)
            (>= dsets 10))
       "ttl_aware|flat")
      ((and (> dsets (* dgets 2)) (>= miss-pct 25))
       (if (>= miss-pct 35) "lfu|flat|pin" "lfu|flat"))
      ((and (> dgets (* dsets 3)) (>= miss-pct 15))
       (if (>= miss-pct 35) "lfu|flat|pin" "lru|flat"))
      ((and (>= get-set 90) (>= miss-pct 4))
       "lru|flat")
      ((and (>= miss-pct 35)
            (> ops 60))
       "lfu|flat|pin")
      (else ""))))
